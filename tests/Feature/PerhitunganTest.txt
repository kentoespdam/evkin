<?php

namespace Tests\Feature;

use App\Models\Master\MasterReports;
use Carbon\Carbon;
use Illuminate\Support\Facades\DB;
use Tests\TestCase;

class PerhitunganTest extends TestCase
{
    /**
     * Test calculation of report formulas with transaction inputs.
     */
    public function test_can_calculate_report_formulas_with_transaction_inputs(): void
    {
        $currentDate = Carbon::create(2026, 1, 1)->startOfMonth();
        $previousMonth = $currentDate->copy()->subMonth();

        echo sprintf(
            "Current Date: %s, Previous Month: %s\n",
            $currentDate->toDateString(),
            $previousMonth->toDateString()
        );

        $masterReports = MasterReports::all();
        $this->assertGreaterThan(0, $masterReports->count(), 'No master reports found');
        // print_r($masterReports->toArray());

        $transactionInputs = DB::table('transaksi_inputs AS ti')
            ->join('master_inputs AS mi', 'ti.master_input_id', '=', 'mi.id')
            ->select('mi.kode', 'ti.periode', 'ti.nilai', 'ti.month', 'ti.year')
            ->whereBetween('ti.periode', [
                $previousMonth->format('Y-m-d'),
                $currentDate->format('Y-m-d'),
            ])
            ->get();

        // print_r($transactionInputs->toArray());
        $this->assertGreaterThan(0, $transactionInputs->count(), 'No transaction inputs found');

        $groupedInputs = $transactionInputs->groupBy('periode');
        $currentPeriodInputs = $groupedInputs->get($currentDate->format('Y-m-d'), collect());
        $previousPeriodInputs = $groupedInputs->get($previousMonth->format('Y-m-d'), collect());

        $currentPeriodeData = array_reduce($currentPeriodInputs->toArray(), function ($carry, $item) {
            $carry[$item->kode] = $item->nilai;
            return $carry;
        }, []);

        $previousPeriodeData = array_reduce($previousPeriodInputs->toArray(), function ($carry, $item) {
            $carry[$item->kode] = $item->nilai;
            return $carry;
        }, []);

        $perhitungan_data = [];
        foreach ($masterReports as $report) {
            $formulaValue = $report->formula;

            $formulaValue = $this->replaceKodeWithNilai(
                $formulaValue,
                $currentPeriodeData
            );

            $this->assertNotEmpty($formulaValue, "Formula value should not be empty for report {$report->descIndicator}");
            $nilaiReport = $this->evaluateFormula($formulaValue);

            $perhitungan_data[] = [
                'master_report_id' => $report->id,
                'periode' => $currentDate->format('Y-m-d'),
                'year' => $currentDate->format('Y'),
                'month' => $currentDate->format('m'),
                'descIndicator' => $report->descIndicator,
                'formula' => $report->formula,
                'formula_value' => $formulaValue,
                'nilai' => round($nilaiReport, 2),
            ];

        }
        print_r($perhitungan_data);
    }

    private function replaceKodeWithNilai($formula, $currentPeriodeData)
    {
        return strtr($formula, $currentPeriodeData);
    }

    private function evaluateFormula($formula)
    {
        try {
            return eval ("return {$formula};");
        } catch (\Throwable $e) {
            return 0;
            // $this->fail("Error evaluating formula: {$formula}. Error: " . $e->getMessage());
        }
    }
}
